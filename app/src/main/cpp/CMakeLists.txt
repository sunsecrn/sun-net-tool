cmake_minimum_required(VERSION 3.22.1)
project(nettool C)
set(CMAKE_C_STANDARD 11)
add_library(zdtun SHARED zdtun/zdtun.c zdtun/utils.c)
target_compile_definitions(zdtun PRIVATE ANDROID _LITTLE_ENDIAN NO_DEBUG)
target_include_directories(zdtun PUBLIC zdtun)
target_link_libraries(zdtun log)
add_library(nettool_capture SHARED capture.c)
target_link_libraries(nettool_capture zdtun android log)
target_link_options(zdtun PRIVATE "-Wl,-z,max-page-size=16384")
target_link_options(nettool_capture PRIVATE "-Wl,-z,max-page-size=16384")
